    **Step 3: Create FirestoreSeeder.kt** in `data/seed/` package.

    ```kotlin
    class FirestoreSeeder(private val db: FirebaseFirestore) {
        suspend fun seedIfEmpty() {
            val snapshot = db.collection(Collections.PRODUCTS).limit(1).get().await()
            if (snapshot.isEmpty) {
                val batch = db.batch()
                SeedData.products.forEach { product ->
                    val ref = db.collection(Collections.PRODUCTS).document(product._id)
                    batch.set(ref, product)
                }
                SeedData.inventoryItems.forEach { item ->
                    val ref = db.collection(Collections.INVENTORY).document(item._id)
                    batch.set(ref, item)
                }
                batch.commit().await()
            }
        }
    }
    ```

    Uses `Collections.PRODUCTS` and `Collections.INVENTORY` constants (from Plan 01) — NOT hardcoded strings.
    Uses `.await()` from `kotlinx.coroutines.tasks.await`.
    Uses `batch.set()` for initial write (NOT `FieldValue.increment()` — that is Phase 3 delta logic only).
